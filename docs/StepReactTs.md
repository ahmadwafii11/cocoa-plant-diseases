1. git clone https://github.com/ahmadwafii11/cocoa-plant-diseases
2. cd cocoa-plant-diseases
3. npm install -g pnpm
4. pnpm -v
5. pnpm create vite name-project (frontend) --template react-ts@latest
6. pnpm install
7. pnpm add react-router-dom
8. pnpm add -D tailwindcss@latest @tailwindcss/vite@latest
9. pnpm add lucide-react
10. pnpm add @fontsource-variable/outfit
11. pnpm add leaflet react-leaflet
12. pnpm add -D @types/leaflet
